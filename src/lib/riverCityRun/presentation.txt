const SIN = Math.sin(.13), COS = Math.cos(.13);
let vehicleYaw=0;
function camera([x,y,z]) { if(vehicleYaw){const dx=x-state.x,cs=Math.cos(vehicleYaw),sn=Math.sin(vehicleYaw);x=state.x+dx*cs+z*sn;z=z*cs-dx*sn;}const yy = y - 8, zz = z + 7; return [x, yy * COS + zz * SIN, zz * COS - yy * SIN]; }
function polygon(points, color, layer = 1) {
  let vertices = points.map(camera), clipped = [];
  for (let i = 0; i < vertices.length; i++) {
    const a = vertices[i], b = vertices[(i + 1) % vertices.length];
    if (a[2] >= .5) clipped.push(a);
    if ((a[2] >= .5) !== (b[2] >= .5)) {
      const t = (.5 - a[2]) / (b[2] - a[2]);
      clipped.push(a.map((v,j) => v + (b[j] - v) * t));
    }
  }
  if (clipped.length < 3) return;
  faces.push({ depth: clipped.reduce((sum,p) => sum + p[2],0) / clipped.length, color, layer,
    points: clipped.map(([x,y,z]) => [width / 2 + x * focal / z, height * .36 - y * focal / z]) });
}
function plane(x1,x2,z1,z2,y,color,layer=0) { polygon([[x1,y,z1],[x2,y,z1],[x2,y,z2],[x1,y,z2]],color,layer); }
function box(x,y,z,w,h,d,colors,layer=1) {

  polygon([[a,t,n],[b,t,n],[b,t,f],[a,t,f]],colors[1],layer);
  if (x >= 0) polygon([[a,y,n],[a,t,n],[a,t,f],[a,y,f]],colors[2],layer);
  else polygon([[b,y,n],[b,y,f],[b,t,f],[b,t,n]],colors[2],layer);
}
